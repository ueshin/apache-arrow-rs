26/// Holds multiple non-contiguous, caller-provided buffers of file data.
27///
28/// This is the in-memory buffer used by the push-based Parquet decoders
29/// (`ParquetPushDecoder` and `ParquetMetaDataPushDecoder`). It can be
30/// constructed up front and handed to a builder so the decoder reuses bytes
31/// that have already been fetched.
32///
33/// Features:
34/// 1. Zero copy
35/// 2. non contiguous ranges of bytes
36///
37/// # Non Coalescing
38///
39/// This buffer does not coalesce  (merging adjacent ranges of bytes into a
40/// single range). Coalescing at this level would require copying the data but
41/// the caller may already have the needed data in a single buffer which would
42/// require no copying.
43///
44/// Thus, the implementation defers to the caller to coalesce subsequent requests
45/// if desired.
46///
47/// # Ordering
48///
49/// The buffers are sorted by the start of their range. Thus, lookups use a
50/// binary search, not a scan of all buffers. Pushed ranges can overlap.
51#[derive(Debug, Clone, Default)]
52pub struct PushBuffers {
53    /// the virtual "offset" of this buffers (added to any request)
54    offset: u64,
55    /// The total length of the file being decoded
56    file_len: u64,
57    /// The ranges of data that are available for decoding (not adjusted for
58    /// offset), sorted by `start`
59    ///
60    /// A `VecDeque`, because a scan pushes buffers at the end and releases
61    /// them at the start. Both are then `O(1)` for each buffer.
62    ranges: VecDeque<Range<u64>>,
63    /// The buffers of data that can be used to decode the Parquet file, in the
64    /// same order as `ranges`
65    buffers: VecDeque<Bytes>,
66    /// The length of the longest range in `ranges`.
67    ///
68    /// This keeps lookups fast in the common case: buffers that do not
69    /// overlap and have similar lengths. Pushed ranges can overlap, so a
70    /// lookup cannot stop at the nearest buffer. `max_len` tells the lookup
71    /// when no earlier buffer can reach the requested range, so it checks one
72    /// or two buffers, not all of them. See [`Self::find`].
73    max_len: u64,
74    /// The number of ranges in `ranges` with the length `max_len`. `max_len`
75    /// can decrease only when this count goes to zero. Thus, a release does
76    /// not scan all ranges to update `max_len` while other ranges of that
77    /// length remain.
78    max_len_count: usize,
79    /// The sum of the lengths of `ranges`, kept up to date so that
80    /// [`Self::buffered_bytes`] does not scan all ranges.
81    buffered_bytes: u64,
82    /// The number of empty ranges in `ranges`. [`Self::push_range`] accepts
83    /// them, and [`Self::release_ranges`] removes them. The count tells
84    /// `release_ranges` when it must look for them in all buffers.
85    empty: usize,
86}

88impl Display for PushBuffers {
89    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
90        writeln!(
91            f,
92            "Buffers (offset: {}, file_len: {})",
93            self.offset, self.file_len
94        )?;
95        writeln!(f, "Available Ranges (w/ offset):")?;
96        for range in &self.ranges {
97            writeln!(
98                f,
99                "  {}..{} ({}..{}): {} bytes",
100                range.start,
101                range.end,
102                range.start + self.offset,
103                range.end + self.offset,
104                range.end - range.start
105            )?;
106        }
107
108        Ok(())
109    }

113    /// Create a new, empty `PushBuffers` for a file of the given length.
114    ///
115    /// Use [`PushBuffers::default`] when the file length is unknown or
116    /// irrelevant (e.g. the push decoder, which tracks ranges by absolute
117    /// offset and never consults `file_len`).
118    pub fn new(file_len: u64) -> Self {
119        Self {
120            offset: 0,
121            file_len,
122            ranges: VecDeque::new(),
123            buffers: VecDeque::new(),
124            max_len: 0,
125            max_len_count: 0,
126            buffered_bytes: 0,
127            empty: 0,
128        }
129    }

131    /// Push all the ranges and buffers
132    ///
133    /// # Errors
134    /// Returns an error if the number of ranges does not match the number of
135    /// buffers, or if any buffer's length does not match its range (see
136    /// [`Self::push_range`]).
137    pub fn push_ranges(
138        &mut self,
139        ranges: Vec<Range<u64>>,
140        buffers: Vec<Bytes>,
141    ) -> Result<(), ParquetError> {
142        if ranges.len() != buffers.len() {
143            return Err(general_err!(
144                "Number of ranges ({}) must match number of buffers ({})",
145                ranges.len(),
146                buffers.len()
147            ));
148        }
149        for (range, buffer) in ranges.into_iter().zip(buffers) {
150            self.push_range(range, buffer)?;
151        }
152        Ok(())
153    }
154
155    /// Push a new range and its associated buffer
156    ///
157    /// # Errors
158    /// Returns an error if the buffer's length does not match the range's
159    /// length, e.g. when a truncated (short) read is pushed.
160    pub fn push_range(&mut self, range: Range<u64>, buffer: Bytes) -> Result<(), ParquetError> {
161        let expected = range.end.saturating_sub(range.start);
162        if expected != buffer.len() as u64 {
163            return Err(general_err!(
164                "Buffer length ({}) does not match length ({}) of range {}..{}",
165                buffer.len(),
166                expected,
167                range.start,
168                range.end
169            ));
170        }
171        // Insert after all buffers that start at or before `range.start`.
172        // Thus, ranges pushed in file order go at the end.
173        let idx = self.ranges.partition_point(|r| r.start <= range.start);
174        match expected.cmp(&self.max_len) {
175            Ordering::Greater => (self.max_len, self.max_len_count) = (expected, 1),
176            Ordering::Equal => self.max_len_count += 1,
177            Ordering::Less => {}
178        }
179        self.buffered_bytes += expected;
180        self.empty += usize::from(expected == 0);
181        self.ranges.insert(idx, range);
182        self.buffers.insert(idx, buffer);
183        Ok(())
184    }

186    /// Returns true if the Buffers contains data for the given range
187    pub(crate) fn has_range(&self, range: &Range<u64>) -> bool {
188        self.find(range.start, range.end).is_some()
189    }

191    /// Returns the index of a buffer that contains all bytes of `start..end`,
192    /// if any.
193    fn find(&self, start: u64, end: u64) -> Option<usize> {
194        // Common case: the buffers do not overlap. Then only the last buffer
195        // that starts at or before `start` can contain `start..end`:
196        //
197        //   buffers:  0..25    ├─────────┤
198        //             25..50             ├─────────┤
199        //             50..75                       ├─────────┤
200        //             75..100                                ├─────────┤
201        //   find:     55..70                         ├─────┤  only 50..75 can contain it
202        //
203        // But pushed buffers can overlap. Then a buffer that starts much
204        // earlier can be the one that contains `start..end`:
205        //
206        //   buffers:  0..100   ├───────────────────────────────────────┤
207        //             50..60                       ├───┤
208        //             55..58                         ├┤
209        //   find:     55..90                         ├─────────────┤  only 0..100 contains it
210        //
211        // Thus, scan back from the last buffer that starts at or before
212        // `start`. Without a limit, a lookup that finds nothing scans all
213        // earlier buffers. `max_len` gives the limit: a buffer that starts
214        // more than `max_len` bytes before `end` ends before `end`, and so do
215        // all buffers before it. Stop there.
216        //
217        // In the common case the scan stops after one or two buffers. It is
218        // long only if a caller pushes one large buffer and then many small
219        // buffers after its start.
220        let candidates = self.ranges.partition_point(|r| r.start <= start);
221        self.ranges
222            .range(..candidates)
223            .enumerate()
224            .rev()
225            .take_while(|(_, r)| r.start.saturating_add(self.max_len) >= end)
226            .find(|(_, r)| r.end >= end)
227            .map(|(idx, _)| idx)
228    }

252    /// Clear any range and corresponding buffer that is exactly in the ranges_to_clear
253    ///
254    /// The cost is `O(log N)` in the number of buffers for each range to
255    /// clear.
256    #[cfg(feature = "arrow")]
257    pub(crate) fn clear_ranges(&mut self, ranges_to_clear: &[Range<u64>]) {
258        // The indexes of the buffers to remove. The buffers are sorted by
259        // start, so all buffers with the same start are adjacent.
260        let mut remove = vec![];
261        for clear in ranges_to_clear {
262            let first = self.ranges.partition_point(|r| r.start < clear.start);
263            let same_start = self
264                .ranges
265                .range(first..)
266                .take_while(|r| r.start == clear.start);
267            remove.extend(
268                same_start
269                    .enumerate()
270                    .filter(|(_, r)| r.end == clear.end)
271                    .map(|(idx, _)| first + idx),
272            );
273        }
274        // Buffers with the same start are not sorted by end, and
275        // `ranges_to_clear` can contain a range more than once.
276        remove.sort_unstable();
277        remove.dedup();
278        let (Some(&first), Some(&last)) = (remove.first(), remove.last()) else {
279            return;
280        };
281        // Move the kept buffers in `first..=last` to the start of that span,
282        // in order, then remove the rest of the span. The buffers outside the
283        // span are not visited.
284        let mut kept = first;
285        let mut remove = remove.into_iter().peekable();
286        for idx in first..=last {
287            if remove.next_if_eq(&idx).is_some() {
288                let len = self.ranges[idx].end.saturating_sub(self.ranges[idx].start);
289                self.buffered_bytes -= len;
290                self.empty -= usize::from(len == 0);
291                if len == self.max_len {
292                    self.max_len_count -= 1;
293                }
294            } else {
295                self.ranges.swap(kept, idx);
296                self.buffers.swap(kept, idx);
297                kept += 1;
298            }
299        }
300        self.ranges.drain(kept..=last);
301        self.buffers.drain(kept..=last);
302        // `max_len` changes only when no buffer of that length remains.
303        if self.max_len_count == 0 {
304            self.update_max_len();
305        }
306    }
307
308    /// Set `max_len` to the maximum length of the remaining ranges, and
309    /// `max_len_count` to the number of ranges of that length.
310    ///
311    /// A `max_len` that is too large is still correct, lookups only scan
312    /// further. This update is for performance: a large buffer that was
313    /// removed must not slow down later lookups.
314    #[cfg(feature = "arrow")]
315    fn update_max_len(&mut self) {
316        (self.max_len, self.max_len_count) = (0, 0);
317        for len in self.ranges.iter().map(|r| r.end.saturating_sub(r.start)) {
318            match len.cmp(&self.max_len) {
319                Ordering::Greater => (self.max_len, self.max_len_count) = (len, 1),
320                Ordering::Equal => self.max_len_count += 1,
321                Ordering::Less => {}
322            }
323        }
324    }

326    /// Remove all buffered bytes in `ranges`.
327    ///
328    /// A buffer that overlaps a range is trimmed or split, and the parts
329    /// outside the range are kept. The kept parts are zero-copy slices, which
330    /// record the bytes that the decoder can still read. Thus, the allocator
331    /// frees the memory of a pushed [`Bytes`] only after all of its parts are
332    /// removed.
333    ///
334    /// The buffers stay sorted by start, which lookups require (see
335    /// [`Self::find`]). If no buffer overlaps `ranges`, the buffers do not
336    /// change.
337    ///
338    /// The cost is `O(log N)` in the number of buffers for each range in
339    /// `ranges`, plus the number of buffers that overlap `ranges`.
340    #[cfg(feature = "arrow")]
341    pub(crate) fn release_ranges(&mut self, ranges: &[Range<u64>]) {
342        let release = merge_ranges(ranges);
343        if release.is_empty() {
344            return;
345        }
346        let overlaps = |range: &Range<u64>| {
347            !range.is_empty() && {
348                let first = release.partition_point(|r| r.end <= range.start);
349                release.get(first).is_some_and(|r| r.start < range.end)
350            }
351        };
352        // Find the buffers that overlap a released range. The buffers are
353        // sorted by start, and no buffer is longer than `max_len`. Thus, a
354        // buffer can overlap `r` only if it starts before `r.end` and less
355        // than `max_len` bytes before `r.start`:
356        //
357        //   buffers:  0..25    ├─────────┤
358        //             25..50             ├─────────┤
359        //             50..75                       ├─────────┤
360        //             75..100                                ├─────────┤
361        //   release:  55..70                         ├─────┤  max_len is 25: check 50..75 only
362        let mut overlapping = vec![];
363        let mut visited = 0;
364        for r in &release {
365            let max_len = self.max_len;
366            let first = self
367                .ranges
368                .partition_point(|b| b.start.saturating_add(max_len) <= r.start);
369            let end = self.ranges.partition_point(|b| b.start < r.end);
370            // A buffer that an earlier released range visited is not visited
371            // again, because `overlaps` checks all released ranges.
372            overlapping
373                .extend((first.max(visited)..end).filter(|&idx| overlaps(&self.ranges[idx])));
374            visited = visited.max(end);
375        }
376        // Most calls release ranges that are no longer buffered. Return
377        // before changing the buffers.
378        let (Some(&first), Some(&last)) = (overlapping.first(), overlapping.last()) else {
379            return;
380        };
381        // Trim the buffers that overlap a released range in place, and do not
382        // clone or move the others. The parts after the first part of a split
383        // buffer go to `split`. A buffer that is released entirely becomes
384        // empty, and is removed below.
385        let mut split = vec![];
386        let mut emptied = false;
387        for idx in overlapping {
388            let (range, buffer) = (&mut self.ranges[idx], &mut self.buffers[idx]);
389            let whole = range.clone();
390            // Each kept part is shorter than `whole`.
391            if whole.end - whole.start == self.max_len {
392                self.max_len_count -= 1;
393            }
394            let offset = |pos: u64| (pos - whole.start) as usize;
395            // The parts of `whole` between the released ranges.
396            let first = release.partition_point(|r| r.end <= whole.start);
397            let mut kept = 0;
398            let mut first_part = None;
399            let mut keep = |part: Range<u64>| {
400                kept += part.end - part.start;
401                if first_part.is_none() {
402                    first_part = Some(part);
403                } else {
404                    let data = buffer.slice(offset(part.start)..offset(part.end));
405                    split.push((part, data));
406                }
407            };
408            let mut start = whole.start;
409            for r in release[first..].iter().take_while(|r| r.start < whole.end) {
410                if start < r.start {
411                    keep(start..r.start);
412                }
413                start = r.end;
414            }
415            if start < whole.end {
416                keep(start..whole.end);
417            }
418            self.buffered_bytes -= (whole.end - whole.start) - kept;
419            match first_part {
420                Some(part) => {
421                    buffer.truncate(offset(part.end));
422                    let _ = buffer.split_to(offset(part.start));
423                    *range = part;
424                }
425                None => {
426                    *buffer = Bytes::new();
427                    *range = whole.start..whole.start;
428                    emptied = true;
429                }
430            }
431        }
432        if emptied || self.empty > 0 {
433            // Remove the released buffers. As before, this also removes empty
434            // buffers that were pushed. The released buffers are all in
435            // `first..=last`, so look at all buffers only if empty buffers
436            // were pushed.
437            let span = match self.empty {
438                0 => first..last + 1,
439                _ => 0..self.ranges.len(),
440            };
441            let mut kept = span.start;
442            for idx in span.clone() {
443                if !self.ranges[idx].is_empty() {
444                    self.ranges.swap(kept, idx);
445                    self.buffers.swap(kept, idx);
446                    kept += 1;
447                }
448            }
449            self.ranges.drain(kept..span.end);
450            self.buffers.drain(kept..span.end);
451            self.empty = 0;
452        }
453        // A trim moves the start of a buffer only past released bytes. Each
454        // other buffer that starts in those bytes is trimmed past them too,
455        // or is empty and removed. Thus, the buffers are still sorted, except
456        // for the split parts, which are appended. Splits are rare, so sort
457        // all buffers then.
458        if !split.is_empty() {
459            let mut parts: Vec<_> = std::mem::take(&mut self.ranges)
460                .into_iter()
461                .zip(std::mem::take(&mut self.buffers))
462                .chain(split)
463                .collect();
464            parts.sort_by_key(|(range, _)| range.start);
465            (self.ranges, self.buffers) = parts.into_iter().unzip();
466        }
467        // `max_len` changes only when no buffer of that length remains. Only
468        // then pay for the pass over all buffers, which keeps `max_len` exact
469        // (see `update_max_len`).
470        if self.max_len_count == 0 {
471            self.update_max_len();
472        }
473    }
474
475    /// Remove all buffered bytes outside `keep`.
476    ///
477    /// This calls [`Self::release_ranges`] with the complement of `keep`, so
478    /// the same rules apply to the kept parts.
479    #[cfg(feature = "arrow")]
480    pub(crate) fn retain_ranges(&mut self, keep: &[Range<u64>]) {
481        let mut release = vec![];
482        let mut start = 0;
483        for range in merge_ranges(keep) {
484            release.push(start..range.start);
485            start = range.end;
486        }
487        release.push(start..u64::MAX);
488        self.release_ranges(&release);
489    }

491    /// Clear all buffered ranges and their corresponding data
492    pub(crate) fn clear_all_ranges(&mut self) {
493        self.ranges.clear();
494        self.buffers.clear();
495        self.max_len = 0;
496        self.max_len_count = 0;
497        self.buffered_bytes = 0;
498        self.empty = 0;
499    }

501    /// Panics if `ranges`, `buffers` and `max_len` do not agree, or if the
502    /// buffers are not sorted.
503    #[cfg(test)]
504    #[track_caller]
505    fn assert_invariants(&self) {
506        assert_eq!(self.ranges.len(), self.buffers.len());
507        assert!(
508            self.ranges.iter().is_sorted_by_key(|r| r.start),
509            "not sorted: {:?}",
510            self.ranges
511        );
512        for (range, buffer) in self.ranges.iter().zip(&self.buffers) {
513            assert_eq!(range.end.saturating_sub(range.start), buffer.len() as u64);
514        }
515        let max_len = self
516            .ranges
517            .iter()
518            .map(|r| r.end.saturating_sub(r.start))
519            .max();
520        let max_len = max_len.unwrap_or(0);
521        assert_eq!(self.max_len, max_len);
522        let is_longest = |r: &&Range<u64>| r.end.saturating_sub(r.start) == max_len;
523        assert_eq!(
524            self.max_len_count,
525            self.ranges.iter().filter(is_longest).count()
526        );
527        let empty = self.ranges.iter().filter(|r| r.is_empty()).count();
528        assert_eq!(self.empty, empty);
529    }

532/// Sort `ranges` by start, remove empty ranges, and merge ranges that
533/// overlap or are adjacent.
534#[cfg(feature = "arrow")]
535fn merge_ranges(ranges: &[Range<u64>]) -> Vec<Range<u64>> {
536    let mut sorted: Vec<Range<u64>> = ranges.iter().filter(|r| !r.is_empty()).cloned().collect();
537    sorted.sort_unstable_by_key(|r| r.start);
538    let mut merged: Vec<Range<u64>> = Vec::with_capacity(sorted.len());
539    for range in sorted {
540        match merged.last_mut() {
541            Some(last) if range.start <= last.end => last.end = last.end.max(range.end),
542            _ => merged.push(range),
543        }
544    }
545    merged
546}

554/// less efficient implementation of Read for Buffers
555impl std::io::Read for PushBuffers {
556    fn read(&mut self, buf: &mut [u8]) -> std::io::Result<usize> {
557        // Find the range that contains the start offset
558        let found = self.find(self.offset, self.offset + buf.len() as u64);
559        if let Some(idx) = found {
560            // Found the range, figure out the starting offset in the buffer
561            let start_offset = (self.offset - self.ranges[idx].start) as usize;
562            let end_offset = start_offset + buf.len();
563            buf.copy_from_slice(&self.buffers[idx][start_offset..end_offset]);
564            // If we found the range, we can return the number of bytes read
565            // advance our offset
566            self.offset += buf.len() as u64;
567            Ok(buf.len())
568        } else {
569            Err(std::io::Error::new(
570                std::io::ErrorKind::UnexpectedEof,
571                "No data available in Buffers",
572            ))
573        }
574    }
575}

577impl ChunkReader for PushBuffers {
578    type T = Self;
579
580    fn get_read(&self, start: u64) -> Result<Self::T, ParquetError> {
581        Ok(self.clone().with_offset(self.offset + start))
582    }
583
584    fn get_bytes(&self, start: u64, length: usize) -> Result<Bytes, ParquetError> {
585        // find the range that contains the start offset
586        if let Some(idx) = self.find(start, start + length as u64) {
587            // Found the range, figure out the starting offset in the buffer
588            let start_offset = (start - self.ranges[idx].start) as usize;
589            return Ok(self.buffers[idx].slice(start_offset..start_offset + length));
590        }
591        // Signal that we need more data
592        let requested_end = start + length as u64;
593        Err(ParquetError::NeedMoreDataRange(start..requested_end))
594    }
595}

760    /// The bytes of a fake file: byte `i` is `i % 251`, so any slice of it is
761    /// easy to build and to check.
762    fn file_bytes(range: Range<u64>) -> Bytes {
763        range.map(|i| (i % 251) as u8).collect::<Vec<u8>>().into()
764    }
765
766    fn push(buffers: &mut PushBuffers, range: Range<u64>) {
767        buffers
768            .push_range(range.clone(), file_bytes(range))
769            .unwrap();
770    }

840    /// Random pushes, clears and lookups, compared with a list that is
841    /// scanned in full for each lookup.
842    #[test]
843    #[cfg(feature = "arrow")]
844    fn fuzz_matches_a_linear_scan() {
845        use rand::rngs::StdRng;
846        use rand::{RngExt, SeedableRng};
847
848        fn random_range(rng: &mut StdRng) -> Range<u64> {
849            let start = rng.random_range(0..500);
850            let len = [0, 1, 5, 20, 100, 300][rng.random_range(0..6)];
851            start..start + len
852        }
853
854        for seed in 0..200 {
855            let mut rng = StdRng::seed_from_u64(seed);
856            let mut buffers = PushBuffers::new(1000);
857            let mut model: Vec<Range<u64>> = vec![];
858            for _ in 0..60 {
859                match rng.random_range(0..10) {
860                    0..4 => {
861                        let range = random_range(&mut rng);
862                        push(&mut buffers, range.clone());
863                        model.push(range);
864                    }
865                    4..6 => {
866                        // Clear some pushed ranges and some other ranges.
867                        let mut clear: Vec<_> = (0..rng.random_range(1..4))
868                            .map(|_| random_range(&mut rng))
869                            .collect();
870                        if !model.is_empty() {
871                            clear.push(model[rng.random_range(0..model.len())].clone());
872                        }
873                        buffers.clear_ranges(&clear);
874                        model.retain(|r| !clear.contains(r));
875                    }
876                    _ => {
877                        let range = random_range(&mut rng);
878                        let expected = model
879                            .iter()
880                            .any(|r| r.start <= range.start && r.end >= range.end);
881                        assert_eq!(buffers.has_range(&range), expected, "seed {seed} {range:?}");
882                        if expected {
883                            let len = (range.end - range.start) as usize;
884                            assert_eq!(
885                                buffers.get_bytes(range.start, len).unwrap(),
886                                file_bytes(range.clone())
887                            );
888                        }
889                    }
890                }
891                assert_valid(&buffers);
892                let mut actual = Vec::from(buffers.ranges.clone());
893                actual.sort_by_key(|r| (r.start, r.end));
894                model.sort_by_key(|r| (r.start, r.end));
895                assert_eq!(actual, model, "seed {seed}");
896            }
897        }
898    }

912    /// The result of `release_ranges`, computed by rebuilding all buffers.
913    #[cfg(feature = "arrow")]
914    fn release_by_rebuild(
915        buffers: &PushBuffers,
916        release: &[Range<u64>],
917    ) -> Vec<(Range<u64>, Bytes)> {
918        let release = merge_ranges(release);
919        let mut parts = vec![];
920        for (range, buffer) in &buffered_parts(buffers) {
921            let mut start = range.start;
922            let mut keep = |part: Range<u64>| {
923                let offset = |pos: u64| (pos - range.start) as usize;
924                parts.push((
925                    part.clone(),
926                    buffer.slice(offset(part.start)..offset(part.end)),
927                ));
928            };
929            for r in release
930                .iter()
931                .filter(|r| r.start < range.end && r.end > range.start)
932            {
933                if start < r.start {
934                    keep(start..r.start);
935                }
936                start = start.max(r.end);
937            }
938            if start < range.end {
939                keep(start..range.end);
940            }
941        }
942        parts.sort_by_key(|(range, _)| range.start);
943        parts
944    }

970    #[test]
971    #[cfg(feature = "arrow")]
972    fn fuzz_release_ranges_matches_rebuild() {
973        use rand::{RngExt, SeedableRng, rngs::StdRng};
974        let data: Vec<u8> = (0..=255).collect();
975        let mut rng = StdRng::seed_from_u64(42);
976        for _ in 0..2000 {
977            let mut buffers = PushBuffers::new(256);
978            for _ in 0..rng.random_range(0..6) {
979                let start = rng.random_range(0..100u64);
980                if rng.random_bool(0.1) {
981                    // An inverted range, which `push_range` accepts.
982                    let end = start.saturating_sub(rng.random_range(1..5u64));
983                    buffers.push_range(start..end, Bytes::new()).unwrap();
984                    continue;
985                }
986                let end = start + rng.random_range(0..20u64);
987                let bytes = Bytes::copy_from_slice(&data[start as usize..end as usize]);
988                buffers.push_range(start..end, bytes).unwrap();
989            }
990            for _ in 0..3 {
991                let release: Vec<_> = (0..rng.random_range(0..4))
992                    .map(|_| {
993                        let start = rng.random_range(0..110u64);
994                        start..start + rng.random_range(0..15u64)
995                    })
996                    .collect();
997                let overlaps = buffers.ranges.iter().any(|range| {
998                    !range.is_empty()
999                        && release
1000                            .iter()
1001                            .any(|r| !r.is_empty() && r.start < range.end && r.end > range.start)
1002                });
1003                let before = buffered_parts(&buffers);
1004                let expected = release_by_rebuild(&buffers, &release);
1005
1006                buffers.release_ranges(&release);
1007                buffers.assert_invariants();
1008                let actual = buffered_parts(&buffers);
1009                if overlaps {
1010                    // The order of buffers with the same start is not
1011                    // specified.
1012                    let by_range = |mut parts: Vec<(Range<u64>, Bytes)>| {
1013                        parts.sort_by_key(|(range, _)| (range.start, range.end));
1014                        parts
1015                    };
1016                    assert_eq!(
1017                        by_range(actual.clone()),
1018                        by_range(expected),
1019                        "release {release:?} of {before:?}"
1020                    );
1021                } else {
1022                    assert_eq!(actual, before, "release {release:?} of {before:?}");
1023                }
1024                for (range, buffer) in actual.iter().filter(|(range, _)| !range.is_empty()) {
1025                    assert_eq!(
1026                        &data[range.start as usize..range.end as usize],
1027                        buffer.as_ref()
1028                    );
1029                }
1030                let total: u64 = actual
1031                    .iter()
1032                    .map(|(r, _)| r.end.saturating_sub(r.start))
1033                    .sum();
1034                assert_eq!(buffers.buffered_bytes(), total);
1035            }
1036        }
